Hook to sign a user in. Call `connectTo` with an `AUTH_CONNECTION` constant. For OAuth and
passwordless connections, the SDK opens the sign-in flow in an in-app browser and returns to your
app through `redirectUrl`. When you pass an `idToken`, the SDK skips the browser and derives the
wallet directly from the token.

### Import

```tsx
import { useWeb3AuthConnect } from '@web3auth/react-native-sdk'
```

```tsx
import { AUTH_CONNECTION, useWeb3AuthConnect } from '@web3auth/react-native-sdk'
import { Button, Text, View } from 'react-native'

function LoginView() {
  const { connectTo, loading, error } = useWeb3AuthConnect()

  return (
    <View>
      <Button
        title={loading ? 'Signing in…' : 'Sign in with Google'}
        disabled={loading}
        onPress={() => connectTo({ authConnection: AUTH_CONNECTION.GOOGLE })}
      />
      {error && <Text>{error.message}</Text>}
    </View>
  )
}
```

```tsx
  const { connectTo, loading } = useWeb3AuthConnect()

  return (
    <View>
      <TextInput
        value={email}
        onChangeText={setEmail}
        placeholder="you@example.com"
        keyboardType="email-address"
        autoCapitalize="none"
      />
      <Button
        title="Continue with email"
        disabled={loading || !email}
        onPress={() =>
          connectTo({
            authConnection: AUTH_CONNECTION.EMAIL_PASSWORDLESS,
            extraLoginOptions: { login_hint: email },
          })
        }
      />
    </View>
  )
}
```

```tsx
import { AUTH_CONNECTION, useWeb3AuthConnect } from '@web3auth/react-native-sdk'
import { Button } from 'react-native'

function FirebaseJwtLoginView() {
  const { connectTo, loading } = useWeb3AuthConnect()

  const login = async () => {
    // Get the JWT from your own auth provider, for example Firebase
    const idToken = await getFirebaseIdToken()

    await connectTo({
      authConnection: AUTH_CONNECTION.CUSTOM,
      authConnectionId: '<YOUR_CONNECTION_ID>', // Connection ID from the dashboard
      idToken,
    })
  }

  return <Button title="Sign in with Firebase" disabled={loading} onPress={login} />
}
```

```tsx
import { AUTH_CONNECTION, useWeb3AuthConnect } from '@web3auth/react-native-sdk'
import { Button, View } from 'react-native'

function GroupedLoginView() {
  const { connectTo, loading } = useWeb3AuthConnect()

  const loginWithGoogle = () =>
    connectTo({
      authConnection: AUTH_CONNECTION.GOOGLE,
      authConnectionId: 'w3a-google',
      groupedAuthConnectionId: 'aggregate-sapphire',
    })

  const loginWithAuth0Email = () =>
    connectTo({
      authConnection: AUTH_CONNECTION.CUSTOM,
      authConnectionId: 'w3a-a0-email-passwordless',
      groupedAuthConnectionId: 'aggregate-sapphire', // same group, same wallet
    })

  return (
    <View>
      <Button title="Google" disabled={loading} onPress={loginWithGoogle} />
      <Button title="Email" disabled={loading} onPress={loginWithAuth0Email} />
    </View>
  )
}
```

### Return type

```tsx
import { type IUseWeb3AuthConnect } from '@web3auth/react-native-sdk'
```

#### `connectTo`

`(params: SdkLoginParams) => Promise<Connection | null>`

Starts sign-in with the specified connection. Resolves with the new
[`connection`](./useWeb3Auth.mdx#connection) on success. On failure, it resolves with `null` and
sets `error` instead of throwing, so check the return value or `error` after you await it.

#### `loading`

`boolean`

`true` while sign-in is in progress.

#### `error`

`Web3authRNError | null`

Error from the most recent `connectTo` call, or `null`.

#### `isConnected`

`boolean`

`true` when the user has an active session. Mirrors `useWeb3Auth().isConnected`.
